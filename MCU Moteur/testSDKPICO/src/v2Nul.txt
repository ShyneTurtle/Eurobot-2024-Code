#include <pico/stdlib.h>
#include <stdio.h>
#include <pico/multicore.h>
#include <hardware/gpio.h>
#include <pico/time.h>
#include <hardware/timer.h>
#include <hardware/clocks.h>
#include <hardware/pio.h>
#include <hardware/structs/systick.h>
#include <hardware/exception.h>

#define FLAG_CORE1 1234

//Mesure de temps
volatile uint32_t period = 0;
volatile uint64_t top = 0;

volatile bool speedUpMode = 0;       //0 -> basse fréquence, 1 -> haute fréquence
                              //Haute fréquence => période inférieur à environ 1ms (1kHz)
                              //Basse fréquence => période suprérieur à environ 1ms (1kHz)

#define MIN_INDEX 5     //Si l'index est trop petit on passe en SLOWED_MODE (correspond à 37Hz)
#define BUFFER_SIZE 10

uint32_t counter64 = 0;
float clkFrequence = 0;    //Fréquence de la clock en Hz
volatile uint32_t buffer[BUFFER_SIZE] = {0};
uint32_t moyenne = 0;
volatile int index = 0;

void core1();

//Interruption signal A
void InterruptSignalA(uint gpio, uint32_t events);

void SysTickINT();

uint64_t GetTime64();

int main()  {
  stdio_init_all();

  set_sys_clock_khz(125000, 0);   //clk_sys à 125MHz

  systick_hw->csr |= 0x00000007;  //Active le compteur de cycle avec l'horloge et les interruptions
  systick_hw->rvr = 0x00ffffff;    //Set la valeur maxdu compteur

  exception_set_exclusive_handler(SYSTICK_EXCEPTION, SysTickINT);

  //LED
  gpio_init(25);
  gpio_set_dir(25, GPIO_OUT);
  
  //Initialise les interruptions sur la broche 21
  gpio_set_irq_enabled_with_callback(21, GPIO_IRQ_EDGE_RISE, true, InterruptSignalA);

  clkFrequence = frequency_count_khz(CLOCKS_FC0_SRC_VALUE_CLK_SYS) * 1000; //clk_cpu en hz;

  //Setup core1 :
  multicore_launch_core1(core1);
  multicore_fifo_pop_blocking(); //Attend le lancement du core1

  //Test :
  gpio_put(25, 1);


  //================== Core0 loop ====================
  while(1)  {
    
    if (multicore_fifo_wready())  {
        multicore_fifo_push_blocking(period);
    }

    //printf("f: %ld\n", period);
  }
}

void core1()  {
  //Envoie un flag lorsqu'il est prêt
  multicore_fifo_push_blocking(FLAG_CORE1);

  //================== Core1 loop ====================
  while(1)  {

    if (multicore_fifo_rvalid())  {
      printf("p: %ld\n", multicore_fifo_pop_blocking());
    }    
  }
}

void InterruptSignalA(uint gpio, uint32_t events) {
  period = GetTime64() - top;  //Enregistre la période
  top = GetTime64();
}

uint64_t GetTime64()  {
  return  (counter64<<24) + (0x00ffffff - systick_hw->cvr);
}

void SysTickINT()   {
  counter64++;
}
